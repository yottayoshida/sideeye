#   apt-file — installs (3.3) and answers --help, but its index is fetched over the network and

#   plocate — installs (1.1.18-1), but its index build chowns the database and fails

mairix	indexes a maildir into a database it can rebuild entirely from the messages beside it	mkdir -p /tmp/md/new /tmp/md/cur /tmp/md/tmp && printf 'From: a@b\nSubject: t\n\nbody\n' > /tmp/md/new/1 && printf 'base=/tmp/md\nmaildir=.\ndatabase=/tmp/md/.index\n' > /tmp/mairixrc && mairix -f /tmp/mairixrc && test -s /tmp/md/.index
fossil	keeps a repository database and a per-checkout state file it recreates on open	mkdir -p /tmp/fw && fossil init /tmp/r.fossil >/dev/null && cd /tmp/fw && fossil open /tmp/r.fossil >/dev/null && test -f /tmp/fw/.fslckout
fontconfig	fc-cache writes per-directory caches that are rebuilt on demand	fc-cache -f /tmp && fc-list >/dev/null
